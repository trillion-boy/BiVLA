\begin{figure}[!t]
  \centering
  \includegraphics[width=\columnwidth]{figure/teaser_bridge_bars.pdf}
  \caption{Success of all 14 configurations on the six SimplerEnv WidowX
  backbones, the original policy counted as one of six tricks. Within a
  trick the bars run from its first to its third setting in the order of
  Section~IV-A, the dotted line marks the original, and a 0 stands where
  a setting succeeded on no episode. No trick raises success by more
  than 10 points on any backbone, while action repeat, depth pruning and
  foveation lower it by 10 to 85 points on several.}
  \label{fig:teaser}

\begin{figure}[t]
  \centering
  \includegraphics[width=\columnwidth]{figure/tradeoff_bridge_col.pdf}
  \caption{Every setting of every trick on the six SimplerEnv WidowX
  backbones, placed by its speedup per environment step and its success
  change from the original policy (the star). Panels read left to right
  and top to bottom as CogACT, CronusVLA, MiniVLA, OpenVLA, SpatialVLA
  and UniVLA, colors identify tricks, and the legend names each setting.
  The shaded corner is faster and no worse than the original. CronusVLA's
  moderate and aggressive reuse settings are omitted because their
  latency is not read (Section~IV-A).}
  \label{fig:tradeoff-widowx}
\end{figure}
