\begin{figure*}[t]
\centering
\begin{tikzpicture}[
    >=Stealth, font=\small,
    box/.style={draw, rounded corners=2pt, minimum height=9mm,
                minimum width=15mm, align=center, inner sep=3pt},
    tag/.style={draw, dashed, rounded corners=2pt, align=center,
                font=\scriptsize, inner sep=3pt, text width=25mm},
    flow/.style={->, thick},
]
% --- pipeline, left to right ---
\node[box] (obs) {Observation\\$o_t$};
\node[box, right=13mm of obs] (enc) {Visual\\encoder};
\node[box, right=20mm of enc] (dec) {Decoder\\layers};
\node[box, right=13mm of dec] (head) {Action\\head};
\node[box, right=15mm of head] (env) {Environment};
\draw[flow] (obs) -- (enc);
\draw[flow] (enc) -- (dec);
\draw[flow] (dec) -- (head);
\draw[flow] (head) -- (env);
% --- feedback arrow above, carrying the horizon ---
\draw[flow] (env.north) -- ++(0,7mm)
      -| node[above, pos=0.25, font=\footnotesize]
      {one call every $h = m\,k$ steps} (obs.north);
% --- the five methods, grouped by where each acts, with its role ---
\node[tag, below=10mm of obs]  (t1) {\textbf{Foveation} (control)};
\node[tag, below=10mm of enc, xshift=17.5mm] (t2) {\textbf{Temporal fusion} (candidate)};
\node[tag, below=10mm of dec]  (t3) {\textbf{Depth pruning} (candidate)};
\node[tag, below=10mm of head] (t4) {\textbf{Action repeat} (control)\\\textbf{Guarded reuse} (candidate)};
\draw[->, dashed] (t1) -- (obs);
\draw[->, dashed] (t2) -- (t2 |- enc.east);
\draw[->, dashed] (t3) -- (dec);
\draw[->, dashed] (t4) -- (head);
\end{tikzpicture}
\caption{The five training-free interventions in the VLA inference pipeline, each
tagged at the point it acts and by its role. The two controls degrade the
input (foveation) or skip feedback unconditionally (action repeat). The
candidates swap reusable visual tokens for the previous step's before they
enter the decoder (temporal fusion), remove decoder layers (depth pruning), or
reuse a whole action under image and action gates (guarded reuse). One call
yields $m$ executed actions and action repeat holds each for $k$ steps, so the horizon $h = m\,k$ sets how many environment steps pass between
calls when no reuse gate skips one.}
\label{fig:pipeline}
\end{figure*}
